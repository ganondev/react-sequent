# Full-Screen Wizard

A **full-screen onboarding wizard** with a persistent progress bar in the chrome, a mix of component and element-returning loaders, and a final `resolve` that returns the collected data.

The first transition uses a normal component loader. The final transition simulates async work before returning JSX with hydrated props. The progress bar updates **immediately** on `advance` — before the async step resolves — because chrome renders outside Suspense.

Click **Start Onboarding** below to try it.

```jsx live
function WizardDemo() {
  const [log, setLog] = React.useState("");

  // --- Singleton provider + hook (shared SequentOutlet) ---
  const WizardContext = React.createContext(null);

  function WizardChrome({ children }) {
    const { context: ctx } = useSequentContext();
    const pct = Math.round((ctx.step / ctx.total) * 100);
    return (
      <div className="demo-wizard">
        <div>
          <div style={{ width: `${pct}%` }} />
        </div>
        <div>Step {ctx.step} of {ctx.total}</div>
        <div>{children}</div>
      </div>
    );
  }

  function WizardProvider({ children }) {
    const { init, status, result, SequentOutlet } = useSequentFlow();

    function openWizard(firstStep, ctx) {
      init(firstStep, ctx);
    }

    return (
      <WizardContext.Provider value={{ openWizard, status, result }}>
        {children}
        <SequentOutlet
          fallback={<span className="demo-fallback">Loading…</span>}
          chrome={(slot) => <WizardChrome>{slot}</WizardChrome>}
        />
      </WizardContext.Provider>
    );
  }

  function useWizard() { return React.useContext(WizardContext); }

  // --- Steps ---
  function StepProfile() {
    const { advance, abort, context } = useSequentStep();
    const [name, setName] = React.useState(context?.data?.name ?? "");
    return (
      <div>
        <h2>Tell us about yourself</h2>
        <p>What should we call you?</p>
        <div>
          <input
            type="text"
            value={name}
            onChange={(e) => setName(e.target.value)}
            placeholder="Your name"
          />
        </div>
        <div>
          <button onClick={() => advance(() => StepPreferences, { step: 2, total: 3, data: { name: name || "User" } })}>
            Continue →
          </button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepPreferences() {
    const { retreat, advance, abort, context } = useSequentStep();
    const { context: ctx } = useSequentContext();
    const [notif, setNotif] = React.useState(context?.data?.notifications ?? true);
    return (
      <div>
        <h2>Set your preferences</h2>
        <p>Communication preferences</p>
        <div>
          <label>
            <input
              type="checkbox"
              checked={notif}
              onChange={(e) => setNotif(e.target.checked)}
            />
            <span>Send me email notifications</span>
          </label>
        </div>
        <div>
          <button onClick={() => retreat()}>← Back</button>
          <button
            onClick={() =>
              advance(async () => {
                const summary = await new Promise((resolve) =>
                  setTimeout(
                    () =>
                      resolve({
                        ...ctx,
                        step: 3,
                        data: { ...ctx.data, notifications: notif },
                      }),
                    700,
                  ),
                );
                return <StepConfirm summary={summary} />;
              }, { ...ctx, step: 3, data: { ...ctx.data, notifications: notif } })
            }
          >
            Continue →
          </button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepConfirm({ summary }) {
    const { retreat, resolve, abort } = useSequentStep();
    return (
      <div>
        <h2>All set! 🎉</h2>
        <p>Your profile is ready.</p>
        <pre>{JSON.stringify(summary, null, 2)}</pre>
        <div>
          <button onClick={() => retreat()}>← Back</button>
          <button onClick={() => resolve(summary.data)}>Finish</button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  // --- Consumer button that uses the singleton hook ---
  function StartOnboarding() {
    const { openWizard, status, result } = useWizard();

    React.useEffect(() => {
      if (status !== "idle" || result === null) return;

      if (result.status === "resolved") {
        setLog(`Onboarding complete: ${JSON.stringify(result.value)}`);
      } else {
        setLog("Onboarding cancelled.");
      }
    }, [status, result]);

    function start() {
      setLog("");
      openWizard(() => StepProfile, { step: 1, total: 3, data: {} });
    }

    return (
      <div className="demo-sandbox">
        <button onClick={start}>Start Onboarding</button>
        {log && <p className="demo-log">{log}</p>}
      </div>
    );
  }

  // --- Root: provider + consumer ---
  return (
    <WizardProvider>
      <StartOnboarding />
    </WizardProvider>
  );
}
```

## Async step loading

The final step waits on a simulated async lookup and then returns JSX with hydrated props:

```js
advance(async () => {
  const summary = await loadWizardSummary();
  return <StepConfirm summary={summary} />;
});
```

`advance(loader, newContext)` accepts a step loader function. That loader can synchronously return a component or asynchronously return a promise. While the loader is pending, the `<Suspense>` fallback inside `WizardChrome` shows the loading indicator.

This uses a timeout to mimic a fetch or server-side decision rather than a lazy import, which is often the more practical case for passing props into the next step.

## Progress via context

The progress bar reads `ctx.step` and `ctx.total` from `useSequentContext()`. Each `advance` call passes updated context:

```js
advance(() => StepPreferences, { step: 2, total: 3, data: { name: "…" } })
```

Because chrome renders **outside** the Suspense boundary, the progress bar jumps from 33% → 66% **immediately** when `advance` is called — before the next step finishes loading.

## Key points

- **Chrome outlives async loads** — the progress bar and step counter stay mounted while the next step resolves.
- **Context still carries shared flow state** — progress metadata stays in context while the final step also receives a hydrated `summary` prop.
- **No centralized step list** — the total step count is just data passed through context; you can change it without touching any config.
