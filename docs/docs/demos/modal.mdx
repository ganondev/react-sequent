# Singleton Modal with Hook

A flow rendered in a **shared singleton modal** — one `SequentOutlet` at the top of the tree inside a provider. Any component below it opens a flow through a single `useModal()` hook, with no refs or outlet rendering of its own.

Click **Open Setup Flow** or **Report a Bug** below — they're unrelated flows sharing the same outlet.

The example also mixes both step-loader styles: a component loader for the first transition, and direct JSX for the last, which merges the final title into context to update the chrome.

Step changes play a **sequential fade-slide transition** via the ready-made `slide({ fade: true })` factory from `react-sequent/transitions` — the outgoing step slides out to the left, then the incoming step slides in from the right.

```jsx live
function SharedModalDemo() {
  // ── Modal infrastructure (would live in its own module in a real app) ─────

  const ModalContext = React.createContext(null);

  function ModalChrome({ children }) {
    const { context: ctx, abort } = useSequentContext();
    return (
      <div className="demo-modal">
        <div>
          <strong>{ctx.title}</strong>
          <button onClick={() => abort("cancelled")}>✕</button>
        </div>
        <div>{children}</div>
      </div>
    );
  }

  function ModalProvider({ children }) {
    const { init, status, result, SequentOutlet } = useSequentFlow();

    return (
      <ModalContext.Provider value={{ openModal: init, status, result }}>
        {children}
        <div style={{ pointerEvents: "auto" }}>
          <SequentOutlet
            transition={slide({ fade: true })}
            chrome={(slot) => (
              <div className="demo-modal-overlay">
                <ModalChrome>{slot}</ModalChrome>
              </div>
            )}
          />
        </div>
      </ModalContext.Provider>
    );
  }

  function useModal() { return React.useContext(ModalContext); }

  // ── Steps ─────────────────────────────────────────────────────────────────

  function StepWelcome() {
    const { advance, abort } = useSequentStep();
    return (
      <div>
        <p>Welcome to the setup flow. This will only take a minute.</p>
        <div>
          <button onClick={() => advance(() => StepDetails, { title: "Details" })}>Get Started →</button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepDetails() {
    const { retreat, advance, abort, context } = useSequentStep();
    const [email, setEmail] = useState(context?.email ?? "");
    return (
      <div>
        <p>Fill in your details below.</p>
        <div>
          <label>Email:</label>
          <input
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="you@example.com"
          />
        </div>
        <div>
          <button onClick={retreat}>← Back</button>
          <button onClick={() => advance(() => <StepDone email={email} />, { title: "All done!", email })}>
            Continue →
          </button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepDone({ email }) {
    const { resolve } = useSequentStep();
    return (
      <div>
        <p>You're all set! 🎉</p>
        <p>Thanks {email}</p>
        <button onClick={() => resolve({ email })}>Close</button>
      </div>
    );
  }

  // ── A second, unrelated flow ─────────────────────────────────────────────

  function StepBugReport() {
    const { advance, abort } = useSequentStep();
    const [description, setDescription] = useState("");
    return (
      <div>
        <p>Describe the issue you encountered.</p>
        <div>
          <textarea
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            placeholder="What went wrong?"
            rows={3}
          />
        </div>
        <div>
          <button
            disabled={!description.trim()}
            onClick={() => advance(() => <StepBugSubmitted description={description} />, { title: "Thanks!", description })}
          >
            Submit →
          </button>
          <button onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepBugSubmitted({ description }) {
    const { resolve } = useSequentStep();
    return (
      <div>
        <p>Bug reported!</p>
        <p>"{description}"</p>
        <button onClick={() => resolve({ submitted: true, description })}>Close</button>
      </div>
    );
  }

  // ── Consumers (pure triggers — no lifecycle awareness) ───────────────────

  function ModalButton({ label, firstStep, ctx }) {
    const { openModal } = useModal();
    return (
      <div className="demo-sandbox">
        <button onClick={() => openModal(firstStep, ctx)}>{label}</button>
      </div>
    );
  }

  // ── Flow observer (decoupled from whichever button started the flow) ─────

  function FlowLogger() {
    const { status, result } = useModal();
    if (status !== "idle") return <p className="demo-log">Flow active — waiting for result…</p>;
    if (result?.status === "resolved") return <p className="demo-log">Flow resolved: {JSON.stringify(result.value)}</p>;
    if (result?.status === "aborted") return <p className="demo-log">Flow aborted.</p>;
  }

  return (
    <ModalProvider>
      <div style={{ display: "flex", gap: "0.5rem" }}>
        <ModalButton label="Open Setup Flow" firstStep={() => StepWelcome} ctx={{ title: "Welcome" }} />
        <ModalButton label="Report a Bug" firstStep={() => StepBugReport} ctx={{ title: "Report a Bug" }} />
      </div>
      <FlowLogger />
    </ModalProvider>
  );
}
```

## Sequential Transition
The demo uses `slide({ fade: true })` from `react-sequent/transitions` — its 24px distance and 300ms duration match the hand-rolled harness this demo used to carry. In sequential mode only one step is visible at a time: while `previousStep` is mounted the exit animation plays and its `onAnimationEnd` fires `onExited`; once the outgoing step unmounts, `nextStep` slides in. `transitionKey` is attached as a React `key` on each animation wrapper so back-to-back queued transitions remount them — otherwise the CSS animations never restart and `onExited` never fires again.

## Outlet Reuse
Both **Setup** and **Report a Bug** are unrelated flows, yet share the same `ModalProvider` and single `SequentOutlet`. Flow state isn't tethered to either button's lifecycle — each `openModal()` call independently activates the outlet, runs to completion (or cancellation), and returns it to idle, ready for the next flow. No per-flow flags are managed by the provider; the library handles that internally.
