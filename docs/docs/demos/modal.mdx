# Singleton Modal with Hook

A flow rendered inside a **shared singleton modal** — a single `SequentOutlet` lives at the top of the tree inside a context provider. This enables an application to enter modal flows with a single shared `useModal` hook: any component in the subtree calls `useModal()` to open a flow without ever touching a ref or rendering an outlet itself.

Click **Open Setup Flow** or **Report a Bug** below to try it — they are unrelated flows sharing the same outlet.

This example mixes both step-loader styles: the first transition uses a component loader, while the last transition returns JSX directly and merges the final title into context to signal chrome updates.

```jsx live
function SharedModalDemo() {
  // ── Modal infrastructure (would live in its own module in a real app) ─────

  const ModalContext = React.createContext(null);

  function ModalChrome({ children }) {
    const { context: ctx, abort } = useSequentContext();
    return (
      <div className="demo-modal">
        <div className="demo-modal__header">
          <strong>{ctx.title}</strong>
          <button className="demo-btn-close" onClick={() => abort("cancelled")}>✕</button>
        </div>
        <div className="demo-modal__body">{children}</div>
      </div>
    );
  }

  function ModalProvider({ children }) {
    const { init, status, result, SequentOutlet } = useSequentFlow();

    function openModal(firstStep, ctx) {
      init(firstStep, ctx);
    }

    return (
      <ModalContext.Provider value={{ openModal, status, result }}>
        {children}
        <div style={{ pointerEvents: "auto" }}>
          <SequentOutlet
            chrome={(slot) => (
              <div className="demo-modal-overlay">
                <ModalChrome>{slot}</ModalChrome>
              </div>
            )}
          />
        </div>
      </ModalContext.Provider>
    );
  }

  function useModal() { return React.useContext(ModalContext); }

  // ── Steps ─────────────────────────────────────────────────────────────────

  function StepWelcome() {
    const { advance, abort } = useSequentStep();
    return (
      <div>
        <p className="demo-para">Welcome to the setup flow. This will only take a minute.</p>
        <div className="demo-actions">
          <button onClick={() => advance(() => StepDetails, { title: "Details" })}>Get Started →</button>
          <button className="demo-btn-ghost" onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepDetails() {
    const { retreat, advance, abort } = useSequentStep();
    const [email, setEmail] = useState("");
    return (
      <div>
        <p className="demo-para">Fill in your details below.</p>
        <div className="demo-input-group">
          <label className="demo-input-label">Email:</label>
          <input
            type="email"
            className="demo-input"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="you@example.com"
          />
        </div>
        <div className="demo-actions">
          <button className="demo-btn-ghost" onClick={retreat}>← Back</button>
          <button onClick={() => advance(() => <StepDone email={email} />, { title: "All done!" })}>
            Continue →
          </button>
          <button className="demo-btn-ghost" onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepDone({ email }) {
    const { resolve } = useSequentStep();
    return (
      <div>
        <p className="demo-para">You're all set! 🎉</p>
        <p className="demo-para">Thanks {email}</p>
        <button onClick={() => resolve({ email })}>Close</button>
      </div>
    );
  }

  // ── A second, unrelated flow ─────────────────────────────────────────────

  function StepBugReport() {
    const { advance, abort } = useSequentStep();
    const [description, setDescription] = useState("");
    return (
      <div>
        <p className="demo-para">Describe the issue you encountered.</p>
        <div className="demo-input-group">
          <textarea
            className="demo-input demo-textarea"
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            placeholder="What went wrong?"
            rows={3}
          />
        </div>
        <div className="demo-actions">
          <button
            disabled={!description.trim()}
            onClick={() => advance(() => <StepBugSubmitted description={description} />, { title: "Thanks!" })}
          >
            Submit →
          </button>
          <button className="demo-btn-ghost" onClick={() => abort("cancelled")}>Cancel</button>
        </div>
      </div>
    );
  }

  function StepBugSubmitted({ description }) {
    const { resolve } = useSequentStep();
    return (
      <div>
        <p className="demo-para">Bug reported!</p>
        <p className="demo-para demo-muted">"{description}"</p>
        <button onClick={() => resolve({ submitted: true })}>Close</button>
      </div>
    );
  }

  // ── Consumers (pure triggers — no lifecycle awareness) ───────────────────

  function SetupButton() {
    const { openModal } = useModal();
    return (
      <div className="demo-sandbox">
        <button onClick={() => openModal(() => StepWelcome, { title: "Welcome" })}>
          Open Setup Flow
        </button>
      </div>
    );
  }

  function ReportBugButton() {
    const { openModal } = useModal();
    return (
      <div className="demo-sandbox">
        <button onClick={() => openModal(() => StepBugReport, { title: "Report a Bug" })}>
          Report a Bug
        </button>
      </div>
    );
  }

  // ── Flow observer (decoupled from whichever button started the flow) ─────

  function FlowLogger() {
    const { status, result } = useModal();

    if (status !== "idle") {
      return <p className="demo-log">Flow active — waiting for result…</p>;
    }

    if (result?.status === "resolved") {
      return <p className="demo-log">Flow resolved: {JSON.stringify(result.value)}</p>;
    }

    if (result?.status === "aborted") { 
      return <p className="demo-log">Flow aborted.</p>;
    }

  }

  return (
    <ModalProvider>
      <div style={{ display: "flex", gap: "0.5rem" }}>
        <SetupButton />
        <ReportBugButton />
      </div>
      <FlowLogger />
    </ModalProvider>
  );
}
```

## Outlet Reuse
Both **Setup** and **Report a Bug** are completely unrelated flows, yet they share the same `ModalProvider` and the same single `SequentOutlet`. The flow state is not tethered to either button's lifecycle — each call to `openModal()` independently activates the outlet, runs to completion (or cancellation), and returns it to idle, ready for the next flow. The provider never manages active or inactive flags per flow; the library handles that internally.
